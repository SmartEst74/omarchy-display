cmake_minimum_required(VERSION 3.27)
project(omarchy-workspace-mirror VERSION 0.1.0 LANGUAGES CXX)
find_package(PkgConfig REQUIRED)
pkg_check_modules(HYPRLAND REQUIRED IMPORTED_TARGET hyprland=0.56.2 libdrm libinput libudev pangocairo pixman-1 wayland-server xkbcommon egl glesv2)
pkg_check_modules(LUA QUIET IMPORTED_TARGET lua5.4)
if(NOT LUA_FOUND)
  pkg_check_modules(LUA REQUIRED IMPORTED_TARGET lua)
endif()
add_library(omarchy-workspace-mirror SHARED src/main.cpp)
target_compile_features(omarchy-workspace-mirror PRIVATE cxx_std_23)
target_compile_definitions(omarchy-workspace-mirror PRIVATE WLR_USE_UNSTABLE)
target_compile_options(omarchy-workspace-mirror PRIVATE -Wall -Wextra -Wpedantic $<$<CXX_COMPILER_ID:GNU>:-fno-gnu-unique>)
target_link_libraries(omarchy-workspace-mirror PRIVATE PkgConfig::HYPRLAND PkgConfig::LUA)
install(TARGETS omarchy-workspace-mirror LIBRARY DESTINATION lib)
